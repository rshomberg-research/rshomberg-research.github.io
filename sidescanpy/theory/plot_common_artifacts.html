Common sidescan artifacts#

Much of what a sidescan image shows is not seabed. The towfish moves, the recorder changes its gain, other sonars ping, the GPS updates slowly and the signal fades into noise at long range. Each leaves its own mark, and an analyst or a detector trained on images sees all of them. This page shows the common ones on public recordings, with what causes each, a number that tells it apart, and what sidescanpy offers for it.

The first figure shows them all side by side. Each section then shows one of them larger, with the artifact boxed and labeled, a zoom where it is small, and, where geometry or navigation is the cause, the same pings placed on a map. Every native image is drawn the same way: ping by slant range, the samples as recorded (the recorder’s own gain, no TVG added) in dB, port on the left. Most of these can only be found, fixed or masked with the raw pings and the aux data, before the image is cut into tiles.

Data: BGS London Gateway (Open Government Licence v3.0, contains data supplied by UKRI), USGS Grand Bay (U.S. public domain), JNCC Wight-Barfleur Reef and mid-Irish Sea (Open Government Licence v3.0); see Example data.

Usage: python examples/theory/plot_common_artifacts.py [output_dir]

    course_over_ground,
    interpolate_aux_onto_pings,
    slice_by_range,
    smooth_navigation,
)

out = Path(sys.argv[1]) if len(sys.argv) > 1 else Path(".")
out.mkdir(parents=True, exist_ok=True)
# The readers warn about known quirks of these files (sample scaling,
# sound velocity); see the source format pages.
configure_logging("ERROR")

MARK = "#00e5ff"  # label colour, chosen to stand out on the copper images


def line(name, freq_key=None):
    """One frequency of a public line, as recorded."""
    survey = read_and_convert([datasets.fetch(name)], fmt="xtf", nav_crs="EPSG:32631")
    return survey, survey.get_frequency(freq_key or survey.frequencies[-1]).dataset


def native(ds, pings, half_range_m):
    """Ping-by-signed-slant-range image in dB (port left), its extent and the sample size."""
    ds = ds.isel(ping=slice(*pings))
    sound = float(np.nanmedian(ds["sound_vel_ms"].values))
    sample_m = (sound if sound > 1000 else 1500.0) / (2 * ds.attrs["sample_rate_hz"])
    n = min(int(half_range_m / sample_m), ds.sizes["sample"])
    amp = np.abs(ds["backscatter_intensity"].values[:, :, :n].astype(float))
    db = 20 * np.log10(np.concatenate([amp[:, 0, ::-1], amp[:, 1]], axis=1) + 1.0)
    first = int(ds["ping"].values[0])
    return db, [-n * sample_m, n * sample_m, first, first + db.shape[0]], sample_m


def show(ax, image, extent, title, xlabel="slant range (m), port left", clim=None):
    ax.set_facecolor("0.85")  # samples with no value (no ground range) show grey
    lo, hi = clim if clim is not None else np.nanpercentile(image, [1, 99.5])
    ax.imshow(image, cmap="copper", vmin=lo, vmax=hi, origin="lower", aspect="auto", extent=extent,
              interpolation="antialiased")
    ax.set_title(title, fontsize=10)
    ax.set_xlabel(xlabel)
    return lo, hi


def side_levels(db, n_side, near, far, sample_m):
    """Per-ping median dB of each side between ``near`` and ``far`` metres of slant range."""
    a, b = int(near / sample_m), int(far / sample_m)
    port = np.median(db[:, n_side - b:n_side - a], axis=1)
    stbd = np.median(db[:, n_side + a:n_side + b], axis=1)
    return port, stbd


def mark(ax, x0, x1, y0, y1, text, inside=False, dashed=False, ha="left"):
    """Box a feature and label it, above the box or just inside its top."""
    ax.add_patch(Rectangle((x0, y0), x1 - x0, y1 - y0, fill=False, ec=MARK, lw=1.6,
                           ls="--" if dashed else "-"))
    if text:
        ax.text(x0 if ha == "left" else x1, y1, text, color=MARK, fontsize=9, fontweight="bold", ha=ha,
                va="top" if inside else "bottom", clip_on=True,
                bbox=dict(fc="black", alpha=0.6, ec="none", pad=1.5))


def zoom(ax, image, extent, clim, box, title, xlabel="slant range (m), port left"):
    """The same image cropped to ``box`` (x0, x1, y0, y1), with the full image's colour scale."""
    show(ax, image, extent, title, xlabel=xlabel, clim=clim)
    ax.set_xlim(box[0], box[1])
    ax.set_ylim(box[2], box[3])


def ground_view(ds, pings, half_range_m, cell_m=0.1):
    """Ping-by-signed-ground-range image in dB: the water column has no ground range."""
    ds = compute_geometric_ranges(altitude_from_bottom_track(ds.isel(ping=slice(*pings)).load()))
    ground = ds["ground_range_m"].values
    db = 20 * np.log10(np.abs(ds["backscatter_intensity"].values.astype(float)) + 1.0)
    axis = np.arange(cell_m / 2, half_range_m, cell_m)
    image = np.full((ds.sizes["ping"], 2, axis.size), np.nan)
    for p in range(ds.sizes["ping"]):
        for c in range(2):
            ok = np.isfinite(ground[p, c])
            if ok.sum() > 1:
                image[p, c] = np.interp(axis, ground[p, c, ok], db[p, c, ok], left=np.nan, right=np.nan)
    first = int(ds["ping"].values[0])
    return (np.concatenate([image[:, 0, ::-1], image[:, 1]], axis=1),
            [-half_range_m, half_range_m, first, first + ds.sizes["ping"]])


def north_up(survey, pings, max_m, *, interpolate=True, course=False, cell_m=0.5, max_hold_s=2.0):
    """Grid a stretch of pings onto a north-up map (local metres from the grid centre).

    ``interpolate`` rebuilds positions between the real fixes; without it
    the positions are used as recorded; ``max_hold_s`` is passed on to
    :func:`~sidescanpy.processing.interpolate_aux_onto_pings`. ``course`` smooths the navigation
    and uses the course over ground as heading, for files with no compass.
    """
    if interpolate:
        survey = interpolate_aux_onto_pings(survey, max_hold_s=max_hold_s)
    if course:
        survey = smooth_navigation(survey, window_s=5.0)
    ds = survey.get_frequency(survey.frequencies[-1]).dataset
    first, last = pings[0], pings[1] if pings[1] is not None else ds.sizes["ping"]
    margin = min(first, 100)  # neighbours for the course at the first pings
    ds = slice_by_range(ds.isel(ping=slice(first - margin, last + 100)), max_m=max_m).load()
    if course:
        ds = ds.assign(heading_deg=("ping", course_over_ground(ds, half_window_s=5.0)))
    ds = ds.isel(ping=slice(margin, margin + last - first))
    ds = with_db_var(compute_georeferencing(compute_geometric_ranges(altitude_from_bottom_track(ds))))
    raster = grid_channel(ds, resolution_m=cell_m, method="mean", backscatter_var="backscatter_db")
    x, y = raster.x_values, raster.y_values
    x0, y0 = float(np.mean(x)), float(np.mean(y))
    east, north = Transformer.from_crs(4326, raster.crs, always_xy=True).transform(
        ds["longitude"].values, ds["latitude"].values)
    half = cell_m / 2
    extent = [x.min() - x0 - half, x.max() - x0 + half, y.min() - y0 - half, y.max() - y0 + half]
    return raster.backscatter, extent, (east - x0, north - y0)


def show_map(ax, grid, extent, track, title):
    lo, hi = np.nanpercentile(grid, [1, 99.5])
    ax.set_facecolor("0.85")
    ax.imshow(grid, cmap="copper", vmin=lo, vmax=hi, origin="lower", extent=extent, interpolation="nearest")
    ax.plot(*track, color=MARK, lw=1.0, ls=":", label="towfish track")
    ax.set_aspect("equal")
    ax.set_xlabel("east (m)")
    ax.set_ylabel("north (m)")
    ax.set_title(title, fontsize=10)

Water column and nadir#

Until the sound reaches the seabed under the towfish there is nothing to echo, so every ping starts with a dark band as wide as the towfish’s altitude. The faint line inside it on the starboard side, at about 10 m, is likely the echo from the sea surface (not confirmed). In ground range, right, the water column has no position: the band and the surface line are gone, and each side starts close to the track. The narrow grey strip left in the middle, about 1 m either side, is ground no sample maps to: there a small step in slant range covers a large step in ground range, so the first seabed samples are spread thin.

What to do: analyze_nadir() finds the first bottom return on every ping, the bottom_track step fills a missing altitude from it, and compute_geometric_ranges() gives the water column no ground range.

altitude = float(np.nanmedian(aisha05["altitude_m"].values[1600:2000]))
print(f"recorded altitude: {altitude:.1f} m")

                                       gridspec_kw=dict(width_ratios=[2, 1, 2]))
clim = show(ax_n, db, extent, "native (slant range)")

Objects in the water column#

The boxed blobs inside the dark band are echoes from things in the water between the towfish and the seabed: fish, a school or debris. They are real echoes, but not seabed. In ground range, and on a map, they are dropped with the rest of the water column; on a tile cut from the native image they look like objects on the bottom.

Far range into noise#

The echo weakens with range by spreading and absorption, and absorption grows with frequency. At about 547 kHz this line’s seabed sinks into the receiver noise long before the 200 m the recorder kept, and the outer part of the swath (boxed) is noise, not dark seabed. The level along range, right, stops falling where the noise takes over.

What to do: estimate_usable_range() finds where the level stops falling (its default, the noise knee); crop or mask beyond it (crop_far_range_noise()).

channel = compute_geometric_ranges(altitude_from_bottom_track(barfleur_h.load()))
usable = analysis.estimate_usable_range(channel, value_var="backscatter_intensity", bin_width_m=2.0)
print(f"usable range {usable.usable_range_m:.0f} m of {float(channel['slant_range_m'].max()):.0f} m recorded")
del channel

db, extent, sample_m = images["Far range into noise"]
n_side = db.shape[1] // 2

for x0, x1, ha in ((extent[0], -u, "left"), (u, extent[1], "right")):

ax_n.set_ylabel("ping")

Towfish motion stripes#

Bright and dark bands cross the whole swath every 20 pings or so, on both sides at once. The towfish is pitching or heaving in the waves, tilting its beams on and off the seabed. Port and starboard brighten and dim together, which seabed would not do; this file records no pitch or heave to confirm it. On the map the stripes stay, because each band is a set of pings, not a place on the seabed. This stretch is slow and slightly curved (the printed track length and course below), so the pings fan out and each band becomes a ray across the swath. This file records no heading, so the map uses the course over ground and no layback; the white streaks on the outer edge of the curve are gaps between pings.

What to do: no correction; know it is there. A detector will see stripes at the wave period.

db, extent, sample_m = images["Towfish motion stripes"]
port, stbd = side_levels(db, db.shape[1] // 2, 20, 60, sample_m)


def detrend(x, window=61):
    return x - np.convolve(np.pad(x, window // 2, mode="edge"), np.ones(window) / window, mode="valid")


r = np.corrcoef(detrend(port), detrend(stbd))[0, 1]
print(f"port and starboard levels, ping to ping: correlation {r:.2f}")

Gain step on one side#

For the first 340 pings the port side is about 8 dB darker than the rest of the line, then it jumps back; starboard does not change. The recorder changed the port receiver’s gain. A few pings, single or in one short run near ping 255, are bright across the port swath (arrows): the same kind of event lasting a ping or a few. Neither is in the seabed. The level of each side along the line, right, shows both.

db, extent, sample_m = images["Gain step on one side"]

Interference from another sonar#

Short bright dashes at a fixed range scatter across both sides, repeating every few pings and stepping in range from one to the next. Another sonar on the same survey (here most likely the sub-bottom profiler) pings out of step with this one, and its pulses arrive at times unrelated to the seabed. The zoom shows the dashes on the dark far range of the starboard side, where nothing else is bright.

What to do: nothing in sidescanpy removes them. A median filter across pings does, but it also smooths small real objects; masking them needs the other sonar’s ping times.

A turn#

The towfish swings through a turn of about 160°. It slows and sinks, so the water column narrows; the inner side is smeared as pings overlap, and the outer side fans apart, leaving gaps between pings at long range. The map shows where that leaves the seabed: the inner swath folds over itself near the centre of the turn and the outer swath is a fan of separate ping lines with empty cells between them. Turns also break the assumptions behind course over ground and a straight-astern layback; with no heading recorded, this map uses the course over ground and no layback, so its inner side is only roughly placed.

heading = np.unwrap(np.radians(aisha16["heading_deg"].values[:675]))
print(f"heading change over the turn: {np.degrees(heading[-1] - heading[0]):+.0f} deg")

db, extent, _ = images["A turn"]
grid, map_extent, track = north_up(aisha16_survey, (0, 675), 75, course=True)
print(f"turn map: {np.isfinite(grid).mean():.0%} of the grid's bounding box has data")
fig, (ax_n, ax_m) = plt.subplots(1, 2, figsize=(15, 7), constrained_layout=True,
                                 gridspec_kw=dict(width_ratios=[1, 1.4]))
show(ax_n, db, extent, "native (slant range)")
ax_n.set_ylabel("ping")

Black streaks on the port side only: runs of samples far below anything the seabed returns, from a recording or telemetry loss (the file does not say which). They are not a very dark seabed; they carry no measurement. Each streak is a few pings tall and runs along range, which the zoom shows.

Navigation that is held, then jumps#

Not visible in the native image, but in every map: on this line the per-ping position stays the same for seconds, then jumps by 20 m or more. Placed by those positions, pings stack on top of each other and then the image tears: the middle map is a row of separate strips, each a pile of pings on one position, with empty gaps between them. interpolate_aux_onto_pings() ramps the positions between real fixes, but by default keeps a value held for more than 2 s (max_hold_s) flat, as it would be for a stop. The towfish was moving at about 6 knots throughout, so these holds are gaps in the GPS, and max_hold_s=20 ramps through them, right. The few slanted single lines in both maps are pings whose recorded heading jumps by about 25 degrees from its neighbours. Navigation: held fixes, interpolation and the track covers navigation in full.

stats = aux_update_events(barfleur_l_survey.navigation)["latitude"]
print(f"navigation record: position changes every {stats.change_gap_s['p50']:.1f} s (median)")

lat = barfleur_l["latitude"].values
changes = np.flatnonzero(np.diff(lat) != 0)
held = np.diff(t[changes]).max() if changes.size > 1 else 0.0

speed = barfleur_l["speed_knots"].values if "speed_knots" in barfleur_l else barfleur_l_survey.navigation["speed_knots"].values
heading_jump = np.abs(np.diff(barfleur_l["heading_deg"].values)).max()
print(f"recorded speed {np.nanmin(speed):.1f} to {np.nanmax(speed):.1f} knots; "
      f"largest ping-to-ping heading change {heading_jump:.0f} deg")

Bedforms and speckle#

Sand waves on this mid-Irish Sea line: each crest is bright on the side facing the sonar and casts a shadow behind it, an object-sized bright and dark pair. Speckle, the grain over everything, comes from many small scatterers adding up with random phases; it is the grain inside every crest and trough, shown closer in the right panel. Both are real seabed, not artifacts, but they make the marks a detector has to learn to ignore.
